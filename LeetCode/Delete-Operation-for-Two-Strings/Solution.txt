2    def minDistance(self, word1: str, word2: str) -> int:

6        for i in range(m + 1):

8                if i == 0 or j == 0 :

11                    L[i][j] = L[i-1][j-1]+1
12                else: 
13                    L[i][j] = max(L[i-1][j], L[i][j-1])
