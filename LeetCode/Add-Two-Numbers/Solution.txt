1# Definition for singly-linked list.
2# class ListNode:
3#     def __init__(self, val=0, next=None):
4#         self.val = val
5#         self.next = next
6
7class Solution:
8    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
9        dummy_head = ListNode(0)
10        current = dummy_head
11        carry = 0
12        while l1 or l2 or carry:
13            val1 = l1.val if l1 else 0
14            val2 = l2.val if l2 else 0
15            total = val1 + val2 + carry
16            carry = total // 10
17            digit = total % 10
18            current.next = ListNode(digit)
19            current = current.next
20            l1 = l1.next if l1 else None
21            l2 = l2.next if l2 else None
22        
23        return dummy_head.next
